import { useState, useEffect } from 'react';
import { ethers } from 'ethers';
import { PublicKey, Connection } from '@solana/web3.js';
import { ConnectButton } from '@rainbow-me/rainbowkit';
import { useAccount, useWalletClient } from 'wagmi';
import { PulseLoader } from 'react-spinners';
import { ArrowRight, Wallet, Zap } from 'lucide-react';
import toast from 'react-hot-toast';
import { ConnectionProvider, WalletProvider, useWallet } from '@solana/wallet-adapter-react';
import { WalletAdapterNetwork } from '@solana/wallet-adapter-base';
import { SalmonWalletAdapter } from '@solana/wallet-adapter-salmon';
import { SnapWalletAdapter } from '@drift-labs/snap-wallet-adapter';
import { clusterApiUrl } from '@solana/web3.js';

const DepositComponent = () => {
  const [destinationSolana, setDestinationSolana] = useState('');
  const [amountEther, setAmountEther] = useState('');
  const [isLoading, setIsLoading] = useState(false);
  const [isVisible, setIsVisible] = useState(false);
  const [manualAddress, setManualAddress] = useState(false);
  const [solBalance, setSolBalance] = useState(null);
  const [transactionHash, setTransactionHash] = useState(null); // State to store transaction hash

  const { address, isConnected } = useAccount();
  const { data: walletClient } = useWalletClient();
  const { select, wallets, publicKey, disconnect } = useWallet();

  useEffect(() => {
    setIsVisible(true);
  }, []);

  useEffect(() => {
    if (publicKey) {
      setDestinationSolana(publicKey.toString());
      fetchSolBalance(publicKey.toString());
    }
  }, [publicKey]);

  const fetchSolBalance = async (solanaAddress) => {
    try {
      const connection = new Connection("https://testnet.dev2.eclipsenetwork.xyz", 'confirmed');
      const balance = await connection.getBalance(new PublicKey(solanaAddress));
      setSolBalance(balance / 1e9); // Convert from lamports to SOL
    } catch (error) {
      console.error('Failed to fetch balance:', error);
    }
  };

  // const contractAddress = '0x83cB71D80078bf670b3EfeC6AD9E5E6407cD0fd1';
  const contractAddress = '0xe49AAa25a10fd6e15DD7DDCb50904cA1E91f6E01';//cli

  const abi = [
    {
      inputs: [
        { internalType: 'bytes32', name: '', type: 'bytes32' },
        { internalType: 'uint256', name: '', type: 'uint256' },
      ],
      name: 'deposit',
      outputs: [],
      stateMutability: 'payable',
      type: 'function',
    },
  ];

  const solanaToBytes32 = (solanaAddress) => {
    try {
      const publicKey = new PublicKey(solanaAddress);
      return ethers.hexlify(publicKey.toBytes().slice(0, 32));
    } catch (error) {
      console.error('Invalid Solana address', error);
      throw new Error('Invalid Solana address');
    }
  };

  const deposit = async () => {
    if (!isConnected || !walletClient) {
      toast.error('Please connect your wallet first!');
      return;
    }

    if (!destinationSolana || !amountEther) {
      toast.error('Please fill in all fields');
      return;
    }

    setIsLoading(true);
    try {
      // Ensure MetaMask is available
      if (typeof window.ethereum === 'undefined') {
        toast.error('MetaMask is not installed');
        return;
      }

      // Request accounts if not already connected
      const accounts = await window.ethereum.request({
        method: 'eth_requestAccounts',
      });

      // If no account is connected, show an error
      if (accounts.length === 0) {
        toast.error('No MetaMask account found');
        return;
      }

      const provider = new ethers.BrowserProvider(window.ethereum);
      const signer = await provider.getSigner();
      const contract = new ethers.Contract(contractAddress, abi, signer);

      const destinationBytes32 = solanaToBytes32(destinationSolana);
      const amountWei = ethers.parseEther(amountEther);

      const tx = await contract.deposit(destinationBytes32, amountWei, { value: amountWei });

      // Set the transaction hash to display it
      setTransactionHash(tx.hash);

      toast.promise(tx.wait(), {
        loading: 'Transaction is pending...',
        success: 'Transaction successful! 🎉',
        error: 'Transaction failed 😥',
      });

      console.log(`Transaction hash: ${tx.hash}`);
    } catch (error) {
      console.error(`Error during deposit: ${error.message}`);
      toast.error(error.message);
    } finally {
      setIsLoading(false);
    }
  };

  const supportedWalletNames = ["Salmon", "Connect by Drift"];
  const supportedWallets = wallets.filter(
    (wallet) =>
      supportedWalletNames.includes(wallet.adapter.name) &&
      (wallet.readyState === "Installed" ||
        wallet.adapter.name === "Connect by Drift")
  );

  return (
    <div className={`max-w-lg mx-auto glass-effect rounded-xl p-8 transition-all duration-500 ${isVisible ? 'opacity-3000 translate-y-0' : 'opacity-0 translate-y-10'} shadow-lg shadow-white/50`}>
      <div className="flex items-center justify-between mb-8 animate-slide-in">
        <div className="flex items-center space-x-3">
          <Zap className="w-8 h-8 text-green-500 animate-pulse-slow" />
          <h1 className="text-3xl font-bold bg-gradient-to-r from-green-400 to-green-600 bg-clip-text text-transparent">
            Eclipse Bridge
          </h1>
        </div>
        <ConnectButton.Custom>
      {({
        account,
        chain,
        openAccountModal,
        openChainModal,
        openConnectModal,
        authenticationStatus,
        mounted,
      }) => {
        const ready = mounted && authenticationStatus !== 'loading';
        const connected =
          ready &&
          account &&
          chain &&
          (!authenticationStatus ||
            authenticationStatus === 'authenticated');

        return (
          <div
            {...(!ready && {
              'aria-hidden': true,
              style: {
                opacity: 0,
                pointerEvents: 'none',
                userSelect: 'none',
              },
            })}
          >
            {(() => {
              if (!connected) {
                return (
                  <button
                    onClick={openConnectModal}
                    type="button"
                    className="px-4 py-2 bg-green-600 text-white rounded-lg hover:bg-green-700 transition duration-300 ease-in-out"
                  >
                    Connect Wallet
                  </button>
                );
              }
              if (chain.unsupported) {
                return (
                  <button
                    onClick={openChainModal}
                    type="button"
                    className="px-4 py-2 bg-red-600 text-white rounded-lg hover:bg-red-700 transition duration-300 ease-in-out"
                  >
                    Wrong network
                  </button>
                );
              }
              return (
                <div style={{ display: 'flex', gap: 12 }}>
                  <button
                    onClick={openChainModal}
                    style={{ display: 'flex', alignItems: 'center' }}
                    type="button"
                    className="px-4 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700 transition duration-300 ease-in-out"
                  >
                    {chain.hasIcon && (
                      <div
                        style={{
                          background: chain.iconBackground,
                          width: 12,
                          height: 12,
                          borderRadius: 999,
                          overflow: 'hidden',
                          marginRight: 4,
                        }}
                      >
                        {chain.iconUrl && (
                          <img
                            alt={chain.name ?? 'Chain icon'}
                            src={chain.iconUrl}
                            style={{ width: 12, height: 12 }}
                          />
                        )}
                      </div>
                    )}
                    {chain.name}
                  </button>
                  <button
                    onClick={openAccountModal}
                    type="button"
                    className="px-4 py-2 bg-green-600 text-white rounded-lg hover:bg-green-700 transition duration-300 ease-in-out"
                  >
                    {account.displayName}
                  </button>
                </div>
              );
            })()}
          </div>
        );
      }}
    </ConnectButton.Custom>
      </div>

      <div className="space-y-6">
        <div className="animate-fade-in" style={{ animationDelay: '0.2s' }}>
          <label className="block text-sm font-medium text-green-400 mb-2">
            Solana Destination Address
          </label>
          {manualAddress ? (
            <input
              type="text"
              className="w-full px-4 py-3 rounded-lg bg-black/50 border border-green-500/30 text-white placeholder-gray-400 focus:ring-2 focus:ring-green-500 focus:border-transparent transition-all duration-300"
              placeholder="Enter Solana address"
              value={destinationSolana}
              onChange={(e) => setDestinationSolana(e.target.value)}
            />
            
          ) : (
            <div className="w-full px-4 py-3 rounded-lg bg-black/50 border border-green-500/30 text-white placeholder-gray-400 focus:ring-2 focus:ring-green-500 focus:border-transparent transition-all duration-300">
              {destinationSolana || 'Connect your Solana wallet'}
            </div>
          )}
          {publicKey && !manualAddress && (
            <div className="mt-2 text-green-400">
              {solBalance !== null ? `${solBalance} SOL` : 'Loading...'}
            </div>
          )}
          <button
            onClick={() => setManualAddress(!manualAddress)}
            className="mt-2 text-green-400 underline"
          >
            {manualAddress ? 'Use connected wallet address' : 'Enter address manually'}
          </button>
        </div>

        <div className="animate-fade-in" style={{ animationDelay: '0.4s' }}>
          <label className="block text-sm font-medium text-green-400 mb-2">
            Amount in ETH
          </label>
          <input
            type="number"
            step="0.000000000000000001"
            className="w-full px-4 py-3 rounded-lg bg-black/50 border border-green-500/30 text-white placeholder-gray-400 focus:ring-2 focus:ring-green-500 focus:border-transparent transition-all duration-300"
            placeholder="0.0"
            value={amountEther}
            onChange={(e) => setAmountEther(e.target.value)}
          />
        </div>

        <button
          onClick={deposit}
          disabled={isLoading || !isConnected}
          className={`w-full py-3 px-6 rounded-lg flex items-center justify-center space-x-2 transition-all duration-300 animate-fade-in
            ${isConnected
              ? 'bg-green-600 hover:bg-green-700 text-white hover:scale-105'
              : 'bg-gray-800 cursor-not-allowed text-gray-500'}`}
          style={{ animationDelay: '0.6s' }}
        >
          {isLoading ? (
            <PulseLoader color="white" size={8} />
          ) : (
            <>
              <span>Bridge Assets</span>
              <ArrowRight className="w-5 h-5" />
            </>
          )}
        </button>
      </div>

      {/* Show transaction hash after successful transaction */}
      {transactionHash && (
        <div className="mt-6 p-4 rounded-lg bg-black/30 border border-green-500/20 animate-fade-in" style={{ animationDelay: '0.8s' }}>
          <div className="flex items-center space-x-2 text-green-400">
            {/* <span>Transaction successful!</span> */}
          </div>
          <div className="mt-2">
            <a
              href={`https://sepolia.etherscan.io/tx/${transactionHash}`}
              target="_blank"
              rel="noopener noreferrer"
              className="text-green-500 underline"
            >
              View Transaction on Explorer
            </a>
          </div>
        </div>
      )}

      {!isConnected && (
        <div className="mt-6 p-4 rounded-lg bg-black/30 border border-green-500/20 animate-fade-in" style={{ animationDelay: '0.8s' }}>
          <div className="flex items-center space-x-2 text-green-400">
            <Wallet className="w-5 h-5" />
            <span>Connect your wallet to start bridging assets</span>
          </div>
        </div>
      )}

      {!publicKey && (
        <div className="mt-6 p-4 rounded-lg bg-black/30 border border-green-500/20 animate-fade-in" style={{ animationDelay: '0.8s' }}>
          <div className="flex items-center space-x-2 text-green-400">
            <Wallet className="w-5 h-5" />
            <span>Connect your Solana wallet</span>
          </div>
          <div className="space-y-4 mt-4">
            {supportedWallets.map((wallet) => (
              <div key={wallet.adapter.name}>
                <button
                  className="w-full bg-indigo-600 hover:bg-indigo-700 text-white font-bold py-3 px-4 rounded-lg transition-colors duration-200 flex items-center justify-center gap-2"
                  onClick={() => select(wallet.adapter.name)}
                >
                  {wallet.adapter.icon && (
                    <img
                      src={wallet.adapter.icon}
                      alt={wallet.adapter.name}
                      className="w-6 h-6"
                    />
                  )}
                  {wallet.adapter.name === "Connect by Drift"
                    ? "MetaMask"
                    : wallet.adapter.name}
                </button>
                <div className="my-2 border-b border-gray-700"></div>
              </div>
            ))}
          </div>
        </div>
      )}
    </div>
  );
};

export default DepositComponent;
